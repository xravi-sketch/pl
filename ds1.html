PROGRAM 01

Array Operations

Design and develop a Java Program to implement Array operations.

Aim

To design and develop a Java program to perform basic array operations such as insertion, deletion, searching, updating and displaying elements.

Java Program
Java 
import java.util.Scanner;

public class ArrayOperations {

    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);

        int[] arr = new int[100];
        int n = 0;
        int choice, pos, value, key;

        while (true) {

            System.out.println("\n----- ARRAY OPERATIONS -----");
            System.out.println("1. Insert");
            System.out.println("2. Delete");
            System.out.println("3. Search");
            System.out.println("4. Update");
            System.out.println("5. Display");
            System.out.println("6. Exit");

            System.out.print("Enter your choice: ");
            choice = sc.nextInt();

            switch (choice) {

                case 1:
                    System.out.print("Enter position (1 to "
                            + (n + 1) + "): ");
                    pos = sc.nextInt();

                    if (pos < 1 || pos > n + 1) {
                        System.out.println("Invalid position!");
                        break;
                    }

                    System.out.print("Enter value: ");
                    value = sc.nextInt();

                    for (int i = n; i >= pos; i--) {
                        arr[i] = arr[i - 1];
                    }

                    arr[pos - 1] = value;
                    n++;

                    System.out.println(
                        "Element inserted successfully."
                    );
                    break;


                case 2:
                    if (n == 0) {
                        System.out.println("Array is empty!");
                        break;
                    }

                    System.out.print(
                        "Enter position to delete (1 to "
                        + n + "): "
                    );

                    pos = sc.nextInt();

                    if (pos < 1 || pos > n) {
                        System.out.println("Invalid position!");
                        break;
                    }

                    for (int i = pos - 1; i < n - 1; i++) {
                        arr[i] = arr[i + 1];
                    }

                    n--;

                    System.out.println(
                        "Element deleted successfully."
                    );
                    break;


                case 3:
                    if (n == 0) {
                        System.out.println("Array is empty!");
                        break;
                    }

                    System.out.print(
                        "Enter element to search: "
                    );

                    key = sc.nextInt();

                    int found = -1;

                    for (int i = 0; i < n; i++) {

                        if (arr[i] == key) {
                            found = i;
                            break;
                        }

                    }

                    if (found != -1) {
                        System.out.println(
                            "Element found at position "
                            + (found + 1)
                        );
                    } else {
                        System.out.println(
                            "Element not found."
                        );
                    }

                    break;


                case 4:
                    if (n == 0) {
                        System.out.println("Array is empty!");
                        break;
                    }

                    System.out.print(
                        "Enter position to update (1 to "
                        + n + "): "
                    );

                    pos = sc.nextInt();

                    if (pos < 1 || pos > n) {
                        System.out.println("Invalid position!");
                        break;
                    }

                    System.out.print("Enter new value: ");
                    value = sc.nextInt();

                    arr[pos - 1] = value;

                    System.out.println(
                        "Element updated successfully."
                    );
                    break;


                case 5:
                    if (n == 0) {
                        System.out.println("Array is empty!");
                        break;
                    }

                    System.out.println("Array elements:");

                    for (int i = 0; i < n; i++) {
                        System.out.print(arr[i] + " ");
                    }

                    System.out.println();
                    break;


                case 6:
                    System.out.println(
                        "Program terminated."
                    );

                    sc.close();
                    return;


                default:
                    System.out.println(
                        "Invalid choice!"
                    );
            }
        }
    }
}
