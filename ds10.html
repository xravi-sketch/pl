PROGRAM 10

Binary Search and Linear Search

Design and develop a Java Program to implement Binary Search and Linear Search.

Aim

To design and develop a Java program to implement Linear Search and Binary Search algorithms and find a given element in an array.

Java Program
Java 
import java.util.Scanner;

public class Search {

    // Linear Search
    static int linearSearch(int[] a, int key) {

        for (int i = 0; i < a.length; i++) {

            if (a[i] == key)
                return i;
        }

        return -1;
    }


    // Binary Search
    static int binarySearch(int[] a, int key) {

        int low = 0;
        int high = a.length - 1;

        while (low <= high) {

            int mid = (low + high) / 2;

            if (a[mid] == key)
                return mid;

            if (key < a[mid])
                high = mid - 1;
            else
                low = mid + 1;
        }

        return -1;
    }


    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);

        System.out.print("Enter number of elements: ");
        int n = sc.nextInt();

        int[] a = new int[n];

        System.out.println("Enter elements in sorted order:");

        for (int i = 0; i < n; i++)
            a[i] = sc.nextInt();

        System.out.print("Enter element to search: ");
        int key = sc.nextInt();


        int linear = linearSearch(a, key);

        if (linear != -1)
            System.out.println(
                "Linear Search: Element found at position "
                + (linear + 1)
            );
        else
            System.out.println(
                "Linear Search: Element not found"
            );


        int binary = binarySearch(a, key);

        if (binary != -1)
            System.out.println(
                "Binary Search: Element found at position "
                + (binary + 1)
            );
        else
            System.out.println(
                "Binary Search: Element not found"
            );

        sc.close();
    }
}
