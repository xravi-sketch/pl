PROGRAM 03

Stack Operations Using Linked List

Design and develop a Java Program to implement Stack operations using Linked List.

Aim

To design and develop a Java program to implement stack operations using a linked list such as push, pop, peek, checking whether the stack is empty, and finding the size of the stack.

Java Program
Java 
import java.util.Scanner;

public class LinkedStack {

    static class Node {
        int data;
        Node next;

        Node(int data) {
            this.data = data;
            this.next = null;
        }
    }

    static Node top = null;
    static int count = 0;

    // Push
    static void push(int data) {

        Node newNode = new Node(data);

        newNode.next = top;
        top = newNode;

        count++;
    }

    // Pop
    static void pop() {

        if (top == null) {
            System.out.println("Stack is empty");
            return;
        }

        System.out.println(
            "Popped element = " + top.data
        );

        top = top.next;
        count--;
    }

    // Peek
    static void peek() {

        if (top == null)
            System.out.println("Stack is empty");
        else
            System.out.println(
                "Top element = " + top.data
            );
    }

    // Check empty
    static void checkEmpty() {

        if (top == null)
            System.out.println("Stack is empty");
        else
            System.out.println("Stack is not empty");
    }

    // Size
    static void size() {

        System.out.println(
            "Stack size = " + count
        );
    }

    // Display
    static void display() {

        Node temp = top;

        System.out.print("Stack = ");

        while (temp != null) {

            System.out.print(temp.data);

            if (temp.next != null)
                System.out.print(" ");

            temp = temp.next;
        }

        System.out.println();
    }

    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);

        char ch;

        do {

            System.out.println("\nLinked Stack Test");

            System.out.println("\nLinked Stack Operations");

            System.out.println("1. push");
            System.out.println("2. pop");
            System.out.println("3. peek");
            System.out.println("4. check empty");
            System.out.println("5. size");

            System.out.print("Enter your Choice: ");

            int choice = sc.nextInt();

            switch (choice) {

                case 1:

                    System.out.print(
                        "Enter integer element to push: "
                    );

                    int value = sc.nextInt();

                    push(value);
                    display();

                    break;

                case 2:

                    pop();
                    display();

                    break;

                case 3:

                    peek();

                    break;

                case 4:

                    checkEmpty();

                    break;

                case 5:

                    size();

                    break;

                default:

                    System.out.println("Invalid choice");
            }

            System.out.print(
                "Do you want to continue (Type y or n): "
            );

            ch = sc.next().charAt(0);

        } while (ch == 'y' || ch == 'Y');

        sc.close();
    }
}
